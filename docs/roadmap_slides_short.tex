\begin{frame}{Phase 1 --- DCL exhausted}

\begin{itemize}
  \item Classifier on best actions.  No value function, no discounting.
  \vfill
  \item \good{Fixed point:} FIFO base $\to$ FIFO's own state distribution $\to$
        deviations never look good $\to$ FIFO clone
  \vfill
  \item Stochastic base, $\varepsilon$-greedy wrapper: not enough
  \vfill
  \item No convergence guarantees
\end{itemize}

\end{frame}

\begin{frame}{Phase 3 --- queue lateness: the failure inverts}

\begin{center}

extraction gap & \good{gone} \\
returns & \good{heavy-tailed}, $\Theta(B^2)$ per tick \\
\midrule

\end{itemize}

\end{frame}

\begin{frame}{Phase 4 --- re-presentation forgiveness}

\[
  \Prob(\text{served in time}) = 1-(1-p)^{m+1}
  \qquad p = 0.4,\; m = 8 \;\Rightarrow\; \good{0.99}
\]

\vfill
\centering
stochastic policy fine $\;\Rightarrow\;$ advantage $\approx 0$ $\;\Rightarrow\;$
sign never trained $\;\Rightarrow\;$ \good{argmax fails}

\vfill
\begin{block}{Per-event action space}
\centering
one $(N{+}1)$-way decision per idle capacity unit
\end{block}

\vfill
\centering
\small
FIFO 70.3802 $=$ 70.3802 \quad never-serve 399.9782 $=$ 399.9782 \quad RVI 40.786 $=$ 40.785

\end{frame}

\begin{frame}{Phase 5 --- fraction served late}

\[
  r_4 \;=\; c_n\Big[\mathbf{1}\{W = D+1\} \;+\; \tfrac{\lambda_n}{\gamma}\,\mathbf{1}\{W \ge D+2\}\Big]
\]

\vfill
\begin{alertblock}{Ill-posed without a rule}
rotting a job \good{suppresses its own arrival stream} $\to$ profitable $\to$
learned policies ``beat'' RVI
\end{alertblock}

\vfill
\centering
\good{Escalation:} late jobs served with forced priority \\[4pt]
\small
action masking only --- PPO, DCL, RVI unchanged \\
RVI truncation now bit-identical at $M = 30, 60, 90$

\end{frame}
